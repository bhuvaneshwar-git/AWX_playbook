step 1 : install minikube
curl -LO https://storage.googleapis.com/minikube/releases/latest/minikube-linux-amd64
sudo install minikube-linux-amd64 /usr/local/bin/minikube && rm minikube-linux-amd64

step 2 : install kubectl  
sudo su
curl -LO "https://dl.k8s.io/release/$(curl -L -s https://dl.k8s.io/release/stable.txt)/bin/linux/amd64/kubectl"
install -o root -g root -m 0755 kubectl /usr/local/bin/kubectl

step 3 : deploy awx-opearator
git clone https://github.com/ansible/awx-operator.git
cd awx-operator/
git checkout 2.19.1 # Or whatever the latest version is
export NAMESPACE=awx
make deploy
kubectl create -f awx-demo.yml -n awx

step 4 : verify your pods and services are running below
kubectl get pods -n awx
kubectl get svc -n awx

step 5 : fetch the url 
minikube service awx-demo-service --url -n awx
